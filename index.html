const CAR_URL = 'https://threejs.org/examples/models/gltf/ferrari.glb'; loader.load( CAR_URL, (gltf) => { carro = gltf.scene; carro.traverse((child) => { if (child.isMesh) { child.castShadow = true; child.receiveShadow = true; if (child.name && child.name.toLowerCase().includes('body')) { if (child.material) { child.material = child.material.clone(); child.material.color = new THREE.Color(0xff0000); child.material.metalness = 0.9; child.material.roughness = 0.25; bodyMaterials.push(child.material); } } } }); scene.add(carro); document.getElementById('loading').style.display = 'none'; }, (xhr) => { const pct = Math.round((xhr.loaded / xhr.total) * 100); document.getElementById('loading').textContent = `Carregando... ${pct}%`; }, (err) => { console.error(err); document.getElementById('loading').textContent = 'Erro ao carregar o modelo 😢'; } );
